語法應用範例

下列範例對應上方 SVG：從型別、介面、泛型到併發取消，展示語法如何形成可測試的專案單元。

資料模型 + Interface + Context 
type Job struct {
    ID   string
    Name string
}

type Store interface {
    Save(ctx context.Context, job Job) error
}

func CreateJob(ctx context.Context, store Store, job Job) error {
    if job.Name == "" {
        return fmt.Errorf("name is required")
    }
    return store.Save(ctx, job)
}
泛型資料轉換 
func Map[T, U any](items []T, fn func(T) U) []U {
    out := make([]U, 0, len(items))
    for _, item := range items {
        out = append(out, fn(item))
    }
    return out
}
可取消併發與 select 
func worker(ctx context.Context, jobs <-chan Job) {
    for {
        select {
        case <-ctx.Done():
            return
        case job, ok := <-jobs:
            if !ok {
                return
            }
            _ = process(job)
        }
    }
}
